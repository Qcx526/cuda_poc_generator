def build_huff_codes(bits, vals):
    codes = {}
    code = 0
    k = 0
    for length in range(1, 17):
        for _ in range(bits[length]):
            codes[vals[k]] = (code, length)
            code += 1
            k += 1
        code <<= 1
    return codes


DC_LUMA = build_huff_codes(DC_LUMA_BITS, DC_LUMA_VALS)
DC_CHROMA = build_huff_codes(DC_CHROMA_BITS, DC_CHROMA_VALS)

class BitWriter:
    def __init__(self):
        self.data = bytearray()
        self.acc = 0
        self.nbits = 0

    def put(self, code, length):
        for i in range(length - 1, -1, -1):
            self.acc = (self.acc << 1) | ((code >> i) & 1)
            self.nbits += 1
            if self.nbits == 8:
                b = self.acc & 0xFF
                self.data.append(b)
                if b == 0xFF:
                    self.data.append(0x00)
                self.acc = 0
                self.nbits = 0

    def finish(self):
        if self.nbits:
            pad = 8 - self.nbits
            self.acc = (self.acc << pad) | ((1 << pad) - 1)
            b = self.acc & 0xFF
            self.data.append(b)
            if b == 0xFF:
                self.data.append(0x00)
            self.acc = 0
            self.nbits = 0
        return bytes(self.data)


def marker(code, payload):
    return bytes([0xFF, code]) + (len(payload) + 2).to_bytes(2, "big") + payload


def app0():
    payload = b"JFIF\x00" + bytes([1, 1, 0]) + (1).to_bytes(2, "big") + (1).to_bytes(2, "big") + b"\x00\x00"
    return marker(0xE0, payload)


def dqt():
    q = bytes([1]) * 64
    payload = bytes([0x00]) + q + bytes([0x01]) + q
    return marker(0xDB, payload)


def dht():
    payload = bytearray()
    for tc_th, bits, vals in [
        (0x00, DC_LUMA_BITS, DC_LUMA_VALS),
        (0x01, DC_CHROMA_BITS, DC_CHROMA_VALS),
        (0x10, AC_LUMA_BITS, AC_LUMA_VALS),
        (0x11, AC_CHROMA_BITS, AC_CHROMA_VALS),
    ]:
        payload.append(tc_th)
        payload.extend(bits[1:17])
        payload.extend(vals)
    return marker(0xC4, bytes(payload))


def sof0(width, height, comps):
    payload = bytearray()
    payload.append(8)
    payload.extend(height.to_bytes(2, "big"))
    payload.extend(width.to_bytes(2, "big"))
    payload.append(len(comps))
    for cid, h, v, tq in comps:
        payload.append(cid)
        payload.append((h << 4) | v)
        payload.append(tq)
    return marker(0xC0, bytes(payload))


def sos(comps):
    payload = bytearray()
    payload.append(len(comps))
    for idx, (cid, h, v, tq) in enumerate(comps):
        selector = 0x00 if idx == 0 else 0x11
        payload.append(cid)
        payload.append(selector)
    payload.extend([0, 63, 0])
    return marker(0xDA, bytes(payload))


def entropy_data(width, height, comps):
    max_h = max(c[1] for c in comps)
    max_v = max(c[2] for c in comps)
    mcu_w = 8 * max_h
    mcu_h = 8 * max_v
    mcu_cols = (width + mcu_w - 1) // mcu_w
    mcu_rows = (height + mcu_h - 1) // mcu_h

    dc_l_c, dc_l_n = DC_LUMA[0]
    ac_l_c, ac_l_n = AC_LUMA[0x00]
    dc_c_c, dc_c_n = DC_CHROMA[0]
    ac_c_c, ac_c_n = AC_CHROMA[0x00]

    bw = BitWriter()
    for _ in range(mcu_rows):
        for _ in range(mcu_cols):
            for ci, (cid, h, v, tq) in enumerate(comps):
                for _ in range(h * v):
                    if ci == 0:
                        bw.put(dc_l_c, dc_l_n)
                        bw.put(ac_l_c, ac_l_n)
                    else:
                        bw.put(dc_c_c, dc_c_n)
                        bw.put(ac_c_c, ac_c_n)
    return bw.finish()


def make_jpeg(width, height, comps):
    out = bytearray()
    out.extend(b"\xff\xd8")
    out.extend(app0())
    out.extend(dqt())
    out.extend(sof0(width, height, comps))
    out.extend(dht())
    out.extend(sos(comps))
    out.extend(entropy_data(width, height, comps))
    out.extend(b"\xff\xd9")
    return bytes(out)


def make_gray_420(width, height):
    comps = [
        (1, 2, 2, 0),  # Y, H=2 V=2, quant table 0
        (2, 1, 1, 1),  # Cb, H=1 V=1, quant table 1
        (3, 1, 1, 1),  # Cr, H=1 V=1, quant table 1
    ]
    return make_jpeg(width, height, comps)
